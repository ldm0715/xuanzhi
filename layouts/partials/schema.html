{{- /* JSON-LD 结构化数据：文章页出 BlogPosting，首页出 WebSite。 用 dict + jsonify 生成，不手工拼字符串——手拼很容易在引号、换行、转义上出错， 而 JSON-LD 必须是合法 JSON，出错就是静默失效（爬虫直接忽略）。 参数： page —— 页面上下文 desc —— head.html 已裁剪好的摘要 image —— 社交卡片图 URL */ -}} {{- $p := .page -}} {{- $desc := .desc -}} {{- $image := .image -}} {{- $author := or site.Params.author site.Title -}} {{- $schema := dict -}} {{- if $p.IsPage -}} {{- $schema = dict "@context" "https://schema.org" "@type" "BlogPosting" "headline" ($p.Title | plainify) "author" (dict "@type" "Person" "name" $author) "publisher" (dict "@type" "Organization" "name" site.Title) "mainEntityOfPage" (dict "@type" "WebPage" "@id" $p.Permalink) "url" $p.Permalink "image" $image "description" $desc -}} {{- /* 没写 date 的页面（文档、说明页这类非文章内容）**不输出日期字段**。 零值会序列化成 "0001-01-01T00:00:00Z"——一个看着像真的、实际不存在的 日期，爬虫可能据此判为无效结构化数据。datePublished 与 dateModified 一起收， 只缺一个同样不自洽。 */ -}} {{- if not $p.Date.IsZero -}} {{- $schema = $schema | merge (dict "datePublished" ($p.Date.Format "2006-01-02T15:04:05Z07:00") "dateModified" ($p.Lastmod.Format "2006-01-02T15:04:05Z07:00") ) -}} {{- end -}} {{- else if $p.IsHome -}} {{- $schema = dict "@context" "https://schema.org" "@type" "WebSite" "name" site.Title "url" site.Home.Permalink "description" $desc -}} {{- end -}} {{- with $schema -}} {{- /* 必须 safeJS： {{- end -}}
