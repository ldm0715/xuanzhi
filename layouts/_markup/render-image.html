{{- /* 图片渲染钩子：page bundle 内图片走 Hugo image processing（WebP + srcset + 防抖动尺寸）， 外链图片原样输出。全部 lazy 加载。 每张图再包一层 ：href 指向最大档（1600px WebP）， 图窗（site.js「图片灯箱」段）接管点击；无 JS 时就是「点开看原图」的朴素降级。 注意：$ 是 hook 上下文；with 块内 . 是 page resource。 */ -}} {{- $u := urls.Parse .Destination -}} {{- $alt := .PlainText -}} {{- $title := .Title -}} {{- $cap := or $title $alt -}} {{- if $u.IsAbs -}} {{- else -}} {{- with .Page.Resources.GetMatch $u.Path -}} {{- if eq .ResourceType "image" -}} {{- $main := .Resize (printf "%dx webp q82" (int (math.Min .Width 1600))) -}} {{- $s1 := .Resize (printf "%dx webp q82" (int (math.Min .Width 640))) -}} {{- $s2 := .Resize (printf "%dx webp q82" (int (math.Min .Width 1024))) -}} {{- $img := printf `` $main.RelPermalink $s1.RelPermalink $s1.Width $s2.RelPermalink $s2.Width $main.RelPermalink $main.Width $main.Width $main.Height ($alt | htmlEscape) -}} {{- $link := printf `%s` $main.RelPermalink ($cap | htmlEscape) $img -}} {{- with $title -}} {{- else -}} {{ $link | safeHTML }} {{- end -}} {{- else -}} {{- end -}} {{- else -}} {{- end -}} {{- end -}}
